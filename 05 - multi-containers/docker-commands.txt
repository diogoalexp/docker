---------------------
Create Network
---------------------

docker network create goals-net

---------------------
Run MongoDB Container
---------------------

docker run --name mongodb \
  -e MONGO_INITDB_ROOT_USERNAME=sa \
  -e MONGO_INITDB_ROOT_PASSWORD=123 \
  -v data:/data/db \
  --rm \
  -d \
  --network goals-net \
  mongo

>docker run --name mongodb -v data:/data/db --rm -d --network goals-net -e MONGO_INITDB_ROOT_USERNAME=sa -e MONGO_INITDB_ROOT_PASSWORD=123 mongo

---------------------
Build Node API Image
---------------------

docker build -t goals-node .

---------------------
Run Node API Container
---------------------

docker run --name goals-backend \
  -e MONGODB_USERNAME=sa \
  -e MONGODB_PASSWORD=123 \
  -v logs:/app/logs \
  -v /Users/maximilianschwarzmuller/development/teaching/udemy/docker-complete/backend:/app \
  -v /app/node_modules \
  --rm \
  -d \
  --network goals-net \
  -p 80:80 \
  goals-node

> docker run --name goals-backend -v "C:\workspace\study\Docker\05 - multi-containers\backend:/app"  -v logs:/app/logs  -v /app/node_modules -e MONGODB_USERNAME=sa -e MONGODB_PASSWORD=123  --rm -p 80:80 --network goals-net goals-node

---------------------
Build React SPA Image
---------------------

docker build -t goals-react .

---------------------
Run React SPA Container
---------------------

docker run --name goals-frontend \
  -v /Users/maximilianschwarzmuller/development/teaching/udemy/docker-complete/frontend/src:/app/src \
  --rm \
  -d \
  -p 3000:3000 \
  -it \
  goals-react

> docker run --name goals-frontend -v "C:\workspace\study\Docker\05 - multi-containers\frontend\src:/app/src"  --rm -p 3000:3000 --network goals-net -it goals-react
---------------------
Stop all Containers
---------------------

docker stop mongodb goals-backend goals-frontend
